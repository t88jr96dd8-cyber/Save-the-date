async function startMonitoring() { const location = document.getElementById("location").value; const service = document.getElementById("service").value; const dateFrom = document.getElementById("dateFrom").value; const dateTo = document.getElementById("dateTo").value; const timeFrom = document.getElementById("timeFrom").value || null; const timeTo = document.getElementById("timeTo").value || null; const telegramUserId = tg.initDataUnsafe?.user?.id || null; const data = { telegramChatId: telegramUserId, telegramUserId: telegramUserId, initData: tg.initData, location: location, service: service, dateFrom: dateFrom, dateTo: dateTo, timeFrom: timeFrom, timeTo: timeTo }; console.log("Monitoring request:", data); try { const response = await fetch( "https://save-the-date-production.up.railway.app/api/watch", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(data) } ); const result = await response.json(); console.log("Backend response:", result); if (result.success) { document.getElementById("popup").style.display = "flex"; } else { alert("Fehler: " + (result.error || "Unbekannter Fehler")); } } catch (error) { console.error(error); alert( "Verbindung zum Server fehlgeschlagen." ); } }
